    <script>
        let isExpanded = false;

        // Header Expansion Logic
        function toggleExpand() {
            const header = document.getElementById('nexusHeader');
            const icon = document.getElementById('hamburgerIcon');
            isExpanded = !isExpanded;
            
            if (isExpanded) {
                header.classList.add('header-expanded');
                // Height based on mobile content (approx 320px)
                document.documentElement.style.setProperty('--header-h', '340px');
                icon.style.transform = 'rotate(90deg)';
            } else {
                header.classList.remove('header-expanded');
                document.documentElement.style.setProperty('--header-h', '70px');
                icon.style.transform = 'rotate(0deg)';
            }
        }

        // Theme Toggle
        function toggleTheme() {
            const html = document.documentElement;
            const btn = document.getElementById('themeToggle');
            const isDark = html.getAttribute('data-theme') === 'dark';
            
            if (isDark) {
                html.setAttribute('data-theme', 'light');
                btn.innerText = '☀️';
            } else {
                html.setAttribute('data-theme', 'dark');
                btn.innerText = '🌙';
            }
        }

        // IMDb Parsing
        function handleImdbInput(input) {
            const match = input.value.match(/tt\d{7,10}/i);
            if (match) {
                input.value = match[0].toLowerCase();
                input.classList.add('text-blue-400');
            } else {
                input.classList.remove('text-blue-400');
            }
        }

        // File Logic
        const dropzone = document.getElementById('dropzone');
        const fileInput = document.getElementById('fileInput');
        const fileList = document.getElementById('fileList');
        const fileListContainer = document.getElementById('fileListContainer');

        dropzone.addEventListener('click', () => fileInput.click());
        dropzone.addEventListener('dragover', (e) => { e.preventDefault(); dropzone.classList.add('dragover'); });
        dropzone.addEventListener('dragleave', () => dropzone.classList.remove('dragover'));
        dropzone.addEventListener('drop', async (e) => {
            e.preventDefault();
            dropzone.classList.remove('dragover');
            handleFiles(e.dataTransfer.files);
        });

        fileInput.addEventListener('change', (e) => handleFiles(e.target.files));

        async function handleFiles(files) {
            fileListContainer.classList.remove('hidden');
            for (const file of files) {
                if (file.name.toLowerCase().endsWith('.zip')) {
                    await processZip(file);
                } else {
                    addToFileList(file.name, (file.size / (1024 * 1024)).toFixed(2) + ' MB');
                }
            }
        }

        async function processZip(zipFile) {
            try {
                const zip = await JSZip.loadAsync(zipFile);
                const entries = Object.keys(zip.files).sort();
                for (const path of entries) {
                    if (!zip.files[path].dir) addToFileList(path, "From ZIP");
                }
            } catch (err) { console.error(err); }
        }

        function addToFileList(name, meta) {
            const div = document.createElement('div');
            div.className = 'file-item p-4 flex justify-between items-center';
            div.innerHTML = `
                <div class="flex items-center gap-3">
                    <div class="w-8 h-8 bg-white/5 rounded-lg flex items-center justify-center">📄</div>
                    <div>
                        <div class="file-name text-xs font-bold truncate max-w-[120px] md:max-w-md">${name}</div>
                        <div class="text-[9px] opacity-40 uppercase font-black">${meta}</div>
                    </div>
                </div>
                <button onclick="editFileName(this)" class="px-4 py-2 bg-white/5 border border-white/10 rounded-xl text-[9px] font-black uppercase">Edit</button>
            `;
            fileList.appendChild(div);
        }

        function editFileName(btn) {
            const nameEl = btn.parentElement.querySelector('.file-name');
            const newName = prompt("Rename file:", nameEl.innerText);
            if (newName) nameEl.innerText = newName;
        }

        function clearFiles() {
            fileList.innerHTML = '';
            fileListContainer.classList.add('hidden');
        }

        function setType(btn) {
            btn.parentElement.querySelectorAll('button').forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
        }

        function setLang(btn) {
            btn.parentElement.querySelectorAll('button').forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
        }
    </script>
